! ==========================================================
! PART 2(a): FULL INTEGRATION (SOLID185, KEYOPT(2) = 0)
! Steel strip: L = 200 mm, b = 20 mm, t = 5 mm
! ==========================================================
ET,1,SOLID185
KEYOPT,1,2,0          ! 0 = Full integration
MP,EX,1,200E3
MP,PRXY,1,0.3

! Geometry & Mesh
BLOCK,0,200,0,20,0,5
VCLEAR,ALL
LSEL,S,LENGTH,,200
LESIZE,ALL,,,20,,1,,,0
LSEL,S,LENGTH,,20
LESIZE,ALL,,,2,,1,,,0
LSEL,S,LENGTH,,5
LESIZE,ALL,,,1,,1,,,0
ALLSEL

MSHKEY,1
VMESH,ALL

! Clamped Edge at x = 0 (Applied purely to nodes to avoid solid/FE conflict)
NSEL,S,LOC,X,0
D,ALL,ALL,0
ALLSEL

! Tip Load: Distributed across x = 200 face nodes
NSEL,S,LOC,X,200
*GET,NE,NODE,0,COUNT
F,ALL,FZ,-10/NE
ALLSEL

! Solution & Postprocessing
/SOLU
ANTYPE,STATIC
SOLVE
FINISH

/POST1
NT = NODE(200,10,0)
*GET,WT_Full,NODE,NT,U,Z
*STATUS,WT_Full
FINISH
